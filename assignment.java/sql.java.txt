CREATE DATABASE LibraryDB;

USE LibraryDB;

CREATE TABLE LibraryBooks (
    book_id INT PRIMARY KEY,
    book_name VARCHAR(50),
    author VARCHAR(50)
);

CREATE TABLE BookIssues (
    issue_id INT PRIMARY KEY,
    student_name VARCHAR(50),
    book_id INT
);

-- INNER JOIN
SELECT l.book_name, b.student_name
FROM LibraryBooks l
INNER JOIN BookIssues b
ON l.book_id = b.book_id;

-- LEFT JOIN
SELECT l.book_name, b.student_name
FROM LibraryBooks l
LEFT JOIN BookIssues b
ON l.book_id = b.book_id;

-- RIGHT JOIN
SELECT l.book_name, b.student_name
FROM LibraryBooks l
RIGHT JOIN BookIssues b
ON l.book_id = b.book_id;

-- CROSS JOIN
SELECT l.book_name, b.student_name
FROM LibraryBooks l
CROSS JOIN BookIssues b;
